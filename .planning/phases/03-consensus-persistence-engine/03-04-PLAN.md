src/wire/frame.rs
- #[repr(u8)] enum Lane { Replication = 0, Heartbeat = 1, Snapshot = 2 } with TryFrom<u8>
- struct FrameHeader { kind: u8, flags: u8, lane: u8, req_id: u64 }  (11 bytes on the wire: kind, flags, lane, req_id big-endian)
- struct Frame { header: FrameHeader, body: bytes::Bytes }
- struct FrameCodec (wraps LengthDelimitedCodec; FrameCodec::new(max_frame_length: usize); set_max_frame_length(&mut self, usize) so a server can cap a connection at a few KiB until its Hello is accepted and widen it per lane afterward; implements tokio_util::codec::{Encoder<Frame>, Decoder<Item = Frame>}; usable synchronously on BytesMut as well as through Framed)
- const KIND_HELLO: u8 = 1; const KIND_HELLO_ACK: u8 = 2; const KIND_ERROR: u8 = 3; kinds 0-15 are reserved by the wire module, application kinds start at 16
- enum FrameError { Io, TooLarge { declared, max }, Truncated }

src/wire/hello.rs
- struct Hello { wire_version: u16, cluster_id: u64, node_id: u64, lane: Lane }  fixed 23-byte body: magic b"LCYC", wire_version u16 BE, cluster_id u64 BE, node_id u64 BE, lane u8
- const WIRE_VERSION: u16 = 1; const DEFAULT_HANDSHAKE_TIMEOUT: Duration = 5 s
- async fn client_handshake<T>(framed: &mut Framed<T, FrameCodec>, hello: Hello, timeout: Duration) -> Result<Hello, HandshakeError>   (sends Hello, expects HelloAck carrying the server's Hello)
- async fn server_handshake<T>(framed: &mut Framed<T, FrameCodec>, expected_cluster_id: u64, my_node_id: u64, timeout: Duration) -> Result<Hello, HandshakeError>   (reads the first frame within timeout, validates, replies HelloAck; on mismatch sends one KIND_ERROR frame then returns the error so the caller closes)
- enum HandshakeError { Timeout, BadMagic, VersionMismatch { theirs, ours }, ClusterMismatch { theirs, ours }, UnexpectedFrame, Io, Frame }
